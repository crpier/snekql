"""Metadata needs type validation as well as constructor optionality."""


def good_token() -> str:
    raise NotImplementedError


def bad_token() -> int:
    raise NotImplementedError


class Good[S = database.Pending](LeftModel[S]):
    __row_type__: ClassVar[database.ReadType[Good[database.Row]]]
    count: Annotated[database.Col[int], Default(7)] = Integer()
    token: Annotated[database.Col[str], Factory(good_token)] = Text()


assert_type(Good().count, int)
assert_type(Good().token, str)
