"""Static-only experiments: no declaration function is executed."""

from typing import Any, Literal, dataclass_transform, overload
from snekql import __BACKEND__ as database
from __META_MODULE__ import ModelMeta as BaseMeta
from snekql.model import Table
from snekql.storage import Attr, _UnboundOwner


@overload
def ServerInteger[T](

@overload

@overload

    raise NotImplementedError


class FlatInteger[T](
    Attr[
        Table[database.Pending],
        Table[database.Row],
        _UnboundOwner,
        T,
        T,
        T,
        Any,
        Literal["__BACKEND__"],
    ]
):
    def __init__(self) -> None:
        raise NotImplementedError

    @staticmethod
    def server_default[V](value: V) -> database.GenCol[V]:
        raise NotImplementedError


class BuilderAttr[T](FlatInteger[T]):
    def db_default[V](self, value: V) -> database.GenCol[V]:
        raise NotImplementedError


def ChainInteger[T]() -> BuilderAttr[T]:
    raise NotImplementedError


class IntegerFactory:
    def __call__[T](self) -> database.Col[T]:
        raise NotImplementedError

    def server_default[T](self, value: T) -> database.GenCol[T]:
        raise NotImplementedError


CallableInteger = IntegerFactory()
