"""Completed selects retain their executable result contract."""

from typing import assert_type

from snekql import __BACKEND__ as database


class Entry[S = database.Pending](database.Model[S, "Entry[database.Fetched]"]):
    id: database.Col[int] = database.Integer(primary_key=True)


async def scoped(transaction: database.Transaction) -> list[Entry[database.Fetched]]:
    query: database.Select[Entry[database.Fetched]] = database.select(Entry).all()
    assert_type(await transaction.fetch_all(query), list[Entry[database.Fetched]])
    return await transaction.fetch_all(query)
