"""Left joins infer an optional fetched model on the right."""

from typing import ClassVar

from typing import assert_type

from snekql import __BACKEND__ as database


class Parent[S = database.Pending](database.Model[S]):
    __row_type__: ClassVar[database.ReadType[Parent[database.Row]]]
    id: database.Col[int] = database.Integer(primary_key=True)


class Child[S = database.Pending](database.Model[S]):
    __row_type__: ClassVar[database.ReadType[Child[database.Row]]]
    id: database.Col[int] = database.Integer(primary_key=True)
    parent_id: database.FKCol[Parent, int] = database.ForeignKey(Parent.id)


def joined() -> database.ClosedRead[tuple[Parent[database.Row], Child[database.Row] | None]]:
    return database.ready(database.select(Parent).left_join(
        Child, on=Child.parent_id.references(Parent.id),
    ).all())


async def optional_child(transaction: database.Transaction) -> None:
    inferred = database.select(Parent).left_join(
        Child, on=Child.parent_id.references(Parent.id),
    ).all()
    assert_type(
        await transaction.fetch_all(inferred),
        list[tuple[Parent[database.Row], Child[database.Row] | None]],
    )
    assert_type(
        await transaction.fetch_all(joined()),
        list[tuple[Parent[database.Row], Child[database.Row] | None]],
    )
