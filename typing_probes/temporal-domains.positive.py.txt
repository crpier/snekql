"""Temporal meanings remain nominal through public query composition."""

from datetime import UTC, date, datetime
from typing import ClassVar, assert_type
from pydantic import BaseModel
from snekql import __BACKEND__ as database


class Event[S = database.Pending](database.Model[S]):
    __row_type__: ClassVar[database.ReadType[Event[database.Row]]]
    instant: database.Col[database.UtcDatetime] = database.Text()
    civil: database.Col[database.LocalDatetime] = database.Text()
    zoned: database.Col[database.ZonedDatetime] = database.Text()
    day: database.Col[date] = database.Text()
    optional: database.Col[database.UtcDatetime | None] = database.Text()


class Snapshot(BaseModel):
    instant: database.UtcDatetime


class Role:
    pass


instant = database.UtcDatetime(datetime(2026, 1, 1, tzinfo=UTC))
civil = database.LocalDatetime(datetime(2026, 1, 1))
zoned = database.ZonedDatetime(datetime(2026, 1, 1, tzinfo=UTC))
assert_type(instant.datetime, datetime)
assert_type(civil.datetime, datetime)
assert_type(zoned.datetime, datetime)
row = Event(instant=instant, civil=civil, zoned=zoned, day=date(2026, 1, 1), optional=None)
assert_type(row.instant, database.UtcDatetime)
assert_type(row.civil, database.LocalDatetime)
assert_type(row.optional, database.UtcDatetime | None)
token = Event.instant.label("instant")
named = database.select(Event).project(Snapshot, instant=token)
derived = named.cte(Role, name="derived")
aliased = database.alias(Event, Role, name="aliased")


def instant_column(column: database.ColumnRef[Event, database.UtcDatetime]) -> database.ColumnRef[Event, database.UtcDatetime]:
    return column


async def control(transaction: database.Transaction) -> None:
    assert_type(await transaction.fetch_all(database.select(Event.instant).where(Event.instant.eq(instant))), list[database.UtcDatetime])
    assert_type(await transaction.fetch_all(database.select(Event.civil).where(Event.civil.eq(civil))), list[database.LocalDatetime])
    assert_type(await transaction.fetch_all(database.select(Event.zoned).where(Event.zoned.eq(zoned))), list[database.ZonedDatetime])
    assert_type(await transaction.fetch_all(database.select(Event.day)), list[date])
    assert_type(await transaction.fetch_all(database.select(Event.optional)), list[database.UtcDatetime | None])
    assert_type(await transaction.fetch_one(database.select(Event.instant.min())), database.UtcDatetime | None)
    assert_type(await transaction.fetch_all(database.select(derived.column(token))), list[database.UtcDatetime])
    assert_type(await transaction.fetch_all(database.select(aliased.column(Event.instant))), list[database.UtcDatetime])
    assert_type(await transaction.fetch_all(database.select(Event.instant).where(Event.instant.eq_col(database.scalar(database.select(Event.instant))))), list[database.UtcDatetime])
    assert_type(await transaction.execute(database.update(Event).set(Event.instant.to(instant)).all()), int)
