"""Bare declarations and native query roles preserve whole-model reads."""

from typing import ClassVar, Literal, assert_type
from pydantic import BaseModel
from snekql import __BACKEND__ as database


class Account[State = database.Pending](database.Model[State]):
    __row_type__: ClassVar[database.ReadType[Account[database.Row]]]
    account_id: database.Col[int] = database.Integer(primary_key=True)


class Peer:
    """A second Account occurrence."""


async def control(transaction: database.Transaction) -> None:
    assert_type(await transaction.fetch_all(database.select(Account)), list[Account[database.Row]])
    peer = database.alias(Account, Peer, name="peer")
    assert_type(await transaction.fetch_all(database.select(peer)), list[Account[database.Row]])
    joined = database.select(Account).join(peer, on=Account.account_id.eq_col(peer.column(Account.account_id)))
    assert_type(await transaction.fetch_all(joined), list[tuple[Account[database.Row], Account[database.Row]]])


class Identifier(BaseModel):
    account_id: int


class Active:
    """A named result role."""


class Lookalike:
    """Structural witnesses alone do not make a query source."""

    @classmethod
    def __backend_family_type__(cls) -> Literal["__BACKEND__"]:
        return "__BACKEND__"

    @classmethod
    def __owner_type__(cls) -> type[Account]:
        return Account

    @classmethod
    def __owner_invariant__(cls, owner: Account) -> Account:
        return owner

    @classmethod
    def __row_type__(cls) -> type[Account[database.Row]]:
        return Account[database.Row]


async def cte_control(transaction: database.Transaction) -> None:
    identifier = Account.account_id.label("account_id")
    active = database.select(Account).project(Identifier, account_id=identifier).cte(Active, name="active")
    assert_type(await transaction.fetch_all(database.select(active)), list[Identifier])
    peer = database.alias(active, Peer, name="active_peer")
    query = database.select(Account).join(peer, on=Account.account_id.eq_col(peer.column(identifier)))
    assert_type(await transaction.fetch_all(query), list[tuple[Account[database.Row], Identifier]])
    left = database.select(Account).left_join(peer, on=Account.account_id.eq_col(peer.column(identifier)))
    assert_type(await transaction.fetch_all(left), list[tuple[Account[database.Row], Identifier | None]])


async def mutation_control(transaction: database.Transaction) -> None:
    assert_type(await transaction.execute(database.update(Account).set(Account.account_id.to(2)).all().returning()), list[Account[database.Row]])
