"""Nested-query factories accept ready same-family and correlated inputs."""

from typing import ClassVar, assert_type
from snekql import __BACKEND__ as database
from snekql import __OTHER_BACKEND__ as foreign


class Account[State = database.Pending](database.Model[State]):
    __row_type__: ClassVar[database.ReadType[Account[database.Row]]]
    number: database.Col[int] = database.Integer()


class Invoice[State = database.Pending](database.Model[State]):
    __row_type__: ClassVar[database.ReadType[Invoice[database.Row]]]
    number: database.Col[int] = database.Integer()


class Payment[State = database.Pending](database.Model[State]):
    __row_type__: ClassVar[database.ReadType[Payment[database.Row]]]
    account: database.FKCol[Account, int] = database.Integer()


class Foreign[State = foreign.Pending](foreign.Model[State]):
    __row_type__: ClassVar[foreign.ReadType[Foreign[foreign.Row]]]
    number: foreign.Col[int] = foreign.Integer()


async def control(transaction: database.Transaction) -> None:
    nested = database.select(Invoice.number).where(Invoice.number.eq_col(Account.number))
    present = database.select(Account).where(database.exists(nested))
    absent = database.select(Account).where(database.not_exists(nested))
    value = database.select(Account.number, database.scalar(nested)).all()
    assert_type(await transaction.fetch_all(present), list[Account[database.Row]])
    assert_type(await transaction.fetch_all(absent), list[Account[database.Row]])
    assert_type(await transaction.fetch_all(value), list[tuple[int, int | None]])

async def fk_control(transaction: database.Transaction) -> None:
    query = database.select(Payment.account).where(Payment.account.in_subquery(database.select(Account.number).all()))
    assert_type(await transaction.fetch_all(query), list[int])
