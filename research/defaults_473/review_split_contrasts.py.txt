"""Contrasts and native ty gaps; accepted gaps are observations, not promises."""
from typing import ClassVar, assert_type


class RequiredSQL[S = database.Pending](ReviewModel[S]):
    __row_type__: ClassVar[database.ReadType[RequiredSQL[database.Row]]]
    value: database.Col[bool] = TwinInteger(server_default=False)


RequiredSQL(value=True)
RequiredSQL()  # error: missing-argument
RequiredSQL(value="wrong")  # error: invalid-argument-type


class RequiredGenerated[S = database.Pending](ReviewModel[S]):
    __row_type__: ClassVar[database.ReadType[RequiredGenerated[database.Row]]]
    value: database.GenCol[bool] = TwinInteger(server_default=False)


explicit_pending = RequiredGenerated(value=database.PENDING_GENERATION)
assert_type(explicit_pending.value, bool | database.PENDING_GENERATION)
RequiredGenerated(value=False)
RequiredGenerated()  # error: missing-argument


class Both[S = database.Pending](ReviewModel[S]):
    __row_type__: ClassVar[database.ReadType[Both[database.Row]]]
    value: database.Col[bool] = TwinInteger(default=False, server_default=True)
    label: database.Col[str] = TwinText(default_factory=lambda: "client", server_default="server")


assert_type(Both().value, bool)
assert_type(Both().label, str)
Both(value=True)
Both(value=None)  # error: invalid-argument-type
