from typing import assert_type

from snekql import __BACKEND__ as db

from models import GeneratedState, PythonDefault, RequiredGenerated, RequiredState, SourceLess

assert_type(required_generated.state, str | db.PENDING_GENERATION)


def row_values(
    client_row: PythonDefault[db.Row],
    required_row: RequiredState[db.Row],
    generated_row: GeneratedState[db.Row],
    source_less_row: SourceLess[db.Row],
    required_generated_row: RequiredGenerated[db.Row],
) -> None:
    assert_type(client_row.state, str)
    assert_type(required_row.state, str)
    assert_type(generated_row.state, str)
    assert_type(source_less_row.state, str)
    assert_type(required_generated_row.state, str)
